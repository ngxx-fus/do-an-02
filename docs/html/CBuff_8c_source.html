21static CBuffSize_t _CBuffGetFreeSpace(const CBuff_t *cb) {
22 if (!cb) return 0;
23 if (cb->stat & CBuffFullMask) return 0;
24 if (cb->head >= cb->tail) return cb->size - (cb->head - cb->tail);
25 return cb->tail - cb->head;
26}

86DefaultRet_t CBuffInit(CBuff_t *cb, void *buffer, CBuffSize_t size) {
87 if (!cb || !buffer || size <= 0) return STAT_ERR_INVALID_ARG;
88
89 CBuffEntry();
90
91 // Clear structure first
92 memset(cb, 0, sizeof(CBuff_t));
93
94 cb->buffer = (CBuffByte_t *)buffer;
95 cb->size = size;
96 cb->head = 0;
97 cb->tail = 0;
98
99 // Init Mutex
100 #if (CBUF_THREAD_SAFE_EN == 1)
101 cb->mutex = xSemaphoreCreateMutex();
102 if (cb->mutex == NULL) {
103 CBuffErr("Mutex Init Failed");
104 return STAT_ERR_INIT_FAILED;
105 }
106 #endif
107
108 // Mark as Initialized, Empty and External Buffer
109 cb->stat = CBuffHasInit | CBuffEmptyMask | CBuffIsExtBuf;
110
111 CBuffLog("Init External Buffer: %p, Size: %d", buffer, size);
112 return STAT_OKE;
113}

115CBuff_t* CBuffCreate(CBuffSize_t size) {
116 if (size <= 0) return NULL;
117 CBuffEntry();
118
119 CBuff_t *cb = (CBuff_t *)malloc(sizeof(CBuff_t));
120 if (!cb) {
121 CBuffErr("Malloc Struct Failed");
122 return NULL;
123 }
124
125 void * bufMem = NULL;
126 // 1. Try Allocate Buffer (Priority: PSRAM -> iRAM)
127 #if (CBUF_USE_PSRAM_EN == 1)
128 bufMem = heap_caps_malloc(size, MALLOC_CAP_SPIRAM);
129 #endif
130
131 if (!bufMem) {
132 bufMem = malloc(size);
133 }
134
135 if (!bufMem) {
136 CBuffErr("Malloc Buffer Failed");
137 free(cb);
138 return NULL;
139 }
140
141 // Reuse CBuffInit logic
142 if (CBuffInit(cb, bufMem, size) != STAT_OKE) {
143 if (bufMem) heap_caps_free(bufMem);
144 free(cb);
145 return NULL;
146 }
147
148 // Clear External Buffer flag because we allocated it internally
149 cb->stat &= ~CBuffIsExtBuf;
150
151 CBuffLog("Created CBuff: %p, Size: %d", cb, size);
152 return cb;
153}

155void CBuffDestroy(CBuff_t *cb) {
156 if (cb == NULL) return;
157 CBuffEntry();
158
159 #if (CBUF_THREAD_SAFE_EN == 1)
160 if (cb->mutex) {
161 vSemaphoreDelete((SemaphoreHandle_t)cb->mutex);
162 cb->mutex = NULL;
163 }
164 #endif
165
166 // Only free buffer if it was allocated by CBuffCreate (Not External)
167 if (cb->buffer != NULL && !(cb->stat & CBuffIsExtBuf)) {
168 heap_caps_free(cb->buffer);
169 cb->buffer = NULL;
170 }
171
172 cb->stat = 0; // Clear all flags
173 free(cb);
174 CBuffExit();
175}

177void CBuffReset(CBuff_t *cb) {
178 if (cb) {
179 VALIDATE_INIT_RET_VOID(cb); // Ensure it's valid before lock
180 CBUF_LOCK(cb);
181 cb->head = 0;
182 cb->tail = 0;
183 // Keep Init and ExtBuf flags, reset Empty/Full
184 CBuffStat_t persistentFlags = cb->stat & (CBuffHasInit | CBuffIsExtBuf);
185 cb->stat = persistentFlags | CBuffEmptyMask;
186 CBUF_UNLOCK(cb);
187 }

190CBuffSize_t CBuffWrite(CBuff_t *cb, const void *data, CBuffSize_t bytes) {
191 VALIDATE_ARGS_RET_ZERO(cb, data, bytes);
192 VALIDATE_INIT_RET_ZERO(cb);
193
194 CBUF_LOCK(cb);
195
196 const CBuffByte_t *data_ptr = (const CBuffByte_t *)data;
197 CBuffSize_t free_space = _CBuffGetFreeSpace(cb);
198 CBuffSize_t bytes_to_write = (bytes > free_space) ? free_space : bytes;
199
200 if (bytes_to_write > 0) {
201 cb->stat &= ~CBuffEmptyMask;
202
203 if (cb->head + bytes_to_write > cb->size) {
204 CBuffSize_t first_part = cb->size - cb->head;
205 memcpy(cb->buffer + cb->head, data_ptr, first_part);
206 memcpy(cb->buffer, data_ptr + first_part, bytes_to_write - first_part);
207 cb->head = bytes_to_write - first_part;
208 } else {
209 memcpy(cb->buffer + cb->head, data_ptr, bytes_to_write);
210 cb->head += bytes_to_write;
211 }
212
213 if (cb->head == cb->size) cb->head = 0;
214 if (cb->head == cb->tail) cb->stat |= CBuffFullMask;
215 }
216
217 CBUF_UNLOCK(cb);
218 return bytes_to_write;
219}

252DefaultRet_t CBuffPush(CBuff_t *cb, const void *data, CBuffSize_t size) {
253 VALIDATE_ARGS(cb, data, size);
254 VALIDATE_INIT(cb);
255
256 CBUF_LOCK(cb);
257
258 if (_CBuffGetFreeSpace(cb) < size) {
259 CBUF_UNLOCK(cb);
260 // CBuffErr("Push Overflow: Req %d, Free %d", size, _CBuffGetFreeSpace(cb)); // Optional Log
261 return STAT_ERR_OVERFLOW;
262 }
263
264 cb->stat &= ~CBuffEmptyMask;
265
266 const CBuffByte_t *data_ptr = (const CBuffByte_t *)data;
267 if (cb->head + size > cb->size) {
268 CBuffSize_t first_part = cb->size - cb->head;
269 memcpy(cb->buffer + cb->head, data_ptr, first_part);
270 memcpy(cb->buffer, data_ptr + first_part, size - first_part);
271 cb->head = size - first_part;
272 } else {
273 memcpy(cb->buffer + cb->head, data_ptr, size);
274 cb->head += size;
275 }
276
277 if (cb->head == cb->size) cb->head = 0;
278 if (cb->head == cb->tail) cb->stat |= CBuffFullMask;
279
280 CBUF_UNLOCK(cb);
281 return STAT_OKE;

284DefaultRet_t CBuffPop(CBuff_t *cb, void *data, CBuffSize_t size) {
285 VALIDATE_ARGS(cb, data, size);
286 VALIDATE_INIT(cb);
287
288 CBUF_LOCK(cb);
289
290 if (_CBuffGetDataCount(cb) < size) {
291 CBUF_UNLOCK(cb);
292 return STAT_ERR_UNDERFLOW;
293 }
294
295 cb->stat &= ~CBuffFullMask;
296
297 uint8_t *out_ptr = (uint8_t *)data;
298 if (cb->tail + size > cb->size) {
299 CBuffSize_t first_part = cb->size - cb->tail;
300 memcpy(out_ptr, cb->buffer + cb->tail, first_part);
301 memcpy(out_ptr + first_part, cb->buffer, size - first_part);
302 cb->tail = size - first_part;
303 } else {
304 memcpy(out_ptr, cb->buffer + cb->tail, size);
305 cb->tail += size;
306 }
307
308 if (cb->tail == cb->size) cb->tail = 0;
309 if (cb->tail == cb->head) cb->stat |= CBuffEmptyMask;
310
311 CBUF_UNLOCK(cb);
312 return STAT_OKE;
313}
