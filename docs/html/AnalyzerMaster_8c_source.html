584 while(gpio_get_level(ANALYZER_READER_PIN_READY) == 0) {
585 if ((esp_timer_get_time() - start_time) / 1000 > READY_WAIT_TIMEOUT_MS) {
586 AMLog("[TaskAnalyzerReaderCom] Timeout waiting for READY signal. Retrying...");
587 break;
588 }
589 vTaskDelay(pdMS_TO_TICKS(1)); // Yield to other tasks
590 }
591
592 if (gpio_get_level(ANALYZER_READER_PIN_READY) == 1) {
593 proceed_with_transaction = true;
594 }
595 #else
596 // If READY pin is disabled, always proceed.
597 proceed_with_transaction = true;
598 #endif
599
600 if (proceed_with_transaction) {
601 AMLog("[TaskAnalyzerReaderCom] READY signal detected or bypassed. Starting transaction.");
602
603 // A single transaction that sends a command and then reads data.
604 // This is more efficient and ensures CS stays low during the whole process.
605 spi_transaction_t t;
606 memset(&t, 0, sizeof(t));
607 t.cmd = ANALYZER_CMD_READ_DATA; // Command to send
608 t.length = buffer_size_bytes * 8; // Total bits to READ
609 t.rx_buffer = spi_rx_buffer; // Buffer to store received data
610
611 ret = spi_device_polling_transmit(spi_handle, &t);
612 if (ret == ESP_OK) {
613 // On success, t.rxlength is the number of bits read.
614 size_t bytes_read = t.rxlength / 8;
615 AMLog("[TaskAnalyzerReaderCom] Command 0x%02X sent, read %d bytes from SPI.", (uint8_t)t.cmd, bytes_read);
616
617 // --- Phase 3: Store data in Circular Buffer ---
618 size_t bytes_written = CBuffWrite(data_cbuff, spi_rx_buffer, bytes_read);
619 if (bytes_written < bytes_read) {
620 AMErr("[TaskAnalyzerReaderCom] Circular buffer is full! Discarded %d bytes.", bytes_read - bytes_written);
621 } else {
622 AMLog("[TaskAnalyzerReaderCom] Wrote %d bytes to circular buffer.", bytes_written);
623 }
624 AMLog("[TaskAnalyzerReaderCom] CBuff status: %u / %u bytes used.", CBuffGetDataCount(data_cbuff), data_cbuff->size);
625
626 AMLog("[TaskAnalyzerReaderCom] Data sample: 0x%04X 0x%04X 0x%04X 0x%04X ...",
627 spi_rx_buffer[0], spi_rx_buffer[1], spi_rx_buffer[2], spi_rx_buffer[3]);
628 } else {
629 AMErr("[TaskAnalyzerReaderCom] SPI transaction failed: %s", esp_err_to_name(ret));
630 }
631 }
632
633 // Wait before next cycle
634 vTaskDelay(pdMS_TO_TICKS(MAIN_LOOP_DELAY_MS));
635 }
636}
